local ServerHop = {
    MaxRetries = 3,
    RetryDelay = 2,
    JobIdCache = {}
}

function ServerHop:FindNewServer(placeId)
    -- Get servers from Roblox API
    local servers = {}
    local success, response = pcall(function()
        return HttpService:JSONDecode(game:HttpGet(
            "https://games.roblox.com/v1/games/"..placeId.."/servers/Public?limit=100"
        ))
    end)
    
    if success and response and response.data then
        for _, server in ipairs(response.data) do
            if server.id ~= game.JobId and server.playing < server.maxPlayers then
                table.insert(servers, server)
            end
        end
    end
    
    return servers
end

function ServerHop:HopToBestServer(placeId)
    local servers = self:FindNewServer(placeId)
    
    -- Sort by player count (prefer less populated servers)
    table.sort(servers, function(a, b)
        return a.playing < b.playing
    end)
    
    -- Try to join available servers
    for i = 1, math.min(3, #servers) do
        local success = pcall(function()
            TeleportService:TeleportToPlaceInstance(placeId, servers[i].id)
        end)
        
        if success then
            return true
        end
        task.wait(self.RetryDelay)
    end
    
    -- If all fails, join a fresh server
    TeleportService:Teleport(placeId)
    return false
end

function ServerHop:Execute()
    local placeId = game.PlaceId
    local retries = 0
    
    while retries < self.MaxRetries do
        if self:HopToBestServer(placeId) then
            return
        end
        retries = retries + 1
        task.wait(self.RetryDelay)
    end
    
    warn("Server hop failed after "..self.MaxRetries.." attempts")
end

local function lerp(a, b, t)
    return a + (b - a) * t
end

            positions[step+1] = Vector3.new(
                lerp(startPos.X, endPos.X, t),
                lerp(startPos.Y, endPos.Y, t),
                lerp(startPos.Z, endPos.Z, t)
            )
        end
        
        -- Execute movement
        local startTime = os.clock()
        for step = 1, #positions do
            hrp.CFrame = CFrame.new(positions[step])
            
            local targetTime = startTime + (stepTime * (step-1))
            local remainingWait = targetTime - os.clock()
            if remainingWait > 0 then
                task.wait(math.min(remainingWait, frameBudget))
            end
        end
    end
    
    -- Movement complete - wait 1 second then execute advanced server hop
    task.wait(1)
    ServerHop:Execute()
end

-- Start the process
task.spawn(function()
    settings().Physics.PhysicsEnvironmentalThrottle = Enum.EnviromentalPhysicsThrottle.Disabled
    ultraFastMove()
end)
